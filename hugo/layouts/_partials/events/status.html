{{/* Returns "active" from 24 hours before an event starts until it ends, "past" once it has ended, otherwise "upcoming". An event ends at its endDate, or its date if it has none; a date without a time lasts all day. */}} {{ $end := time (.Params.endDate | default .Date) }} {{ if not (or $end.Hour $end.Minute) }} {{ $end = $end.AddDate 0 0 1 }} {{ end }} {{ $status := "upcoming" }} {{ if lt $end now }} {{ $status = "past" }} {{ else if le .Date (now.Add (duration "hour" 24)) }} {{ $status = "active" }} {{ end }} {{ return $status }}
